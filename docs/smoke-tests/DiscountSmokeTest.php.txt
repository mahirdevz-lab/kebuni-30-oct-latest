<?php

namespace App\Commands;

use App\Services\DailyClosingService;
use App\Services\GoodsReceiptService;
use App\Services\PosReturnService;
use App\Services\PosSaleService;
use App\Services\SupervisorAuthService;
use CodeIgniter\CLI\BaseCommand;
use CodeIgniter\CLI\CLI;

/** TEMPORARY: diskon manual + PIN supervisor. Membersihkan data uji sendiri. */
class DiscountSmokeTest extends BaseCommand
{
    protected $group = 'pos';
    protected $name  = 'pos:discount-smoke';

    private int $fails = 0;

    private function check(string $label, bool $ok, $detail = ''): void
    {
        if (! $ok) {
            $this->fails++;
        }
        CLI::write(($ok ? '[PASS] ' : '[FAIL] ') . $label . ($detail !== '' ? '  => ' . (is_scalar($detail) ? $detail : json_encode($detail)) : ''), $ok ? 'green' : 'red');
    }

    public function run(array $params)
    {
        $db  = db_connect();
        $ids = array_map('intval', array_column($db->table('products')->where('status', 'aktif')->where('deleted_at', null)->orderBy('id')->get(2)->getResultArray(), 'id'));
        [$A, $B] = $ids;
        $sup = (int) $db->table('suppliers')->where('deleted_at', null)->get(1)->getRow('id');
        $db->table('products')->whereIn('id', $ids)->update(['is_pos' => 1]);
        foreach ([$A => 10000, $B => 5000] as $pid => $price) {
            $db->table('pos_product_prices')->insert(['product_id' => $pid, 'retail_price' => $price, 'effective_from' => date('Y-m-d H:i:s', time() - 60), 'created_at' => date('Y-m-d H:i:s')]);
        }
        // user uji: kasir (role kasir), supervisor (role supervisor-toko), admin
        $mk = function (string $username, string $roleSlug, string $pw) use ($db) {
            $rid = (int) $db->table('roles')->where('slug', $roleSlug)->get()->getRow('id');
            $db->table('users')->insert(['role_id' => $rid, 'full_name' => 'Uji ' . $username, 'username' => $username, 'email' => $username . '@example.test', 'password_hash' => password_hash($pw, PASSWORD_DEFAULT), 'status' => 'active', 'created_at' => date('Y-m-d H:i:s'), 'updated_at' => date('Y-m-d H:i:s')]);

            return (int) $db->insertID();
        };
        $kasirId = $mk('uji_kasir', 'kasir', 'pw-kasir-1');
        $spvId   = $mk('uji_spv', 'supervisor-toko', 'pw-spv-1');
        $spv2Id  = $mk('uji_spv2', 'supervisor-toko', 'pw-spv-2');
        $auth    = new SupervisorAuthService();

        // --- PIN
        $this->check('Kasir tidak berwenang menyetel PIN', ! $auth->setPin($kasirId, 'pw-kasir-1', '4821')['success']);
        $this->check('PIN non-digit / pendek ditolak', ! $auth->setPin($spvId, 'pw-spv-1', 'abcd')['success'] && ! $auth->setPin($spvId, 'pw-spv-1', '12')['success']);
        $this->check('PIN mudah ditebak (1234, 1111) ditolak', ! $auth->setPin($spvId, 'pw-spv-1', '1234')['success'] && ! $auth->setPin($spvId, 'pw-spv-1', '1111')['success']);
        $this->check('Password akun salah ditolak', ! $auth->setPin($spvId, 'salah', '4821')['success']);
        $this->check('Supervisor menyetel PIN 4821', $auth->setPin($spvId, 'pw-spv-1', '4821')['success']);
        $stored = $db->table('users')->where('id', $spvId)->get()->getRow('pin_hash');
        $this->check('PIN disimpan sebagai hash (bukan plaintext)', $stored && $stored !== '4821' && password_verify('4821', $stored));
        $this->check('Supervisor 2 belum menyetel PIN → tidak muncul di daftar', in_array($spvId, array_map('intval', array_column($auth->eligible(), 'id')), true) && ! in_array($spv2Id, array_map('intval', array_column($auth->eligible(), 'id')), true));
        $this->check('PIN benar diterima', $auth->verify($spvId, '4821', 'pos.discount')['success']);
        $this->check('Supervisor tanpa PIN ditolak', ! $auth->verify($spv2Id, '4821', 'pos.discount')['success']);
        $this->check('Kasir tidak bisa jadi penyetuju (tanpa PIN)', ! $auth->verify($kasirId, '4821', 'pos.discount')['success']);
        $w = $auth->verify($spvId, '0000', 'pos.discount');
        $this->check('PIN salah → pesan sisa percobaan', ! $w['success'] && str_contains($w['message'], '4 percobaan'), $w['message']);
        $auth->verify($spvId, '0000', 'pos.discount'); $auth->verify($spvId, '0000', 'pos.discount'); $auth->verify($spvId, '0000', 'pos.discount');
        $lk = $auth->verify($spvId, '0000', 'pos.discount');
        $this->check('5× salah → terkunci', ! $lk['success'] && str_contains($lk['message'], 'Terkunci'), $lk['message']);
        $this->check('Saat terkunci PIN benar pun ditolak', ! $auth->verify($spvId, '4821', 'pos.discount')['success']);
        $db->table('users')->where('id', $spvId)->update(['pin_locked_until' => null, 'pin_failed' => 0]);
        $this->check('Setelah kunci habis PIN benar diterima lagi', $auth->verify($spvId, '4821', 'pos.discount')['success']);

        // --- Checkout
        $day = new DailyClosingService(); $day->open(0, 1);
        (new GoodsReceiptService())->create(['supplier_id' => $sup, 'items' => [['product_id' => $A, 'qty_regular' => 5, 'line_cost' => 25000], ['product_id' => $B, 'qty_regular' => 50, 'line_cost' => 100000]]], 1);
        $svc   = new PosSaleService();
        $kasir = ['user_id' => $kasirId, 'can_override' => false, 'can_discount' => false];
        $uuid  = static fn () => sprintf('%04x%04x-%04x-%04x-%04x-%04x%04x%04x', mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff));
        $cart  = [['product_id' => $A, 'qty' => 2], ['product_id' => $B, 'qty' => 4]]; // 20.000 + 20.000 = 40.000
        $pay   = [['method' => 'CASH', 'tendered' => 40000]];
        $disc  = ['type' => 'PCT', 'value' => 10, 'reason' => 'pelanggan langganan'];

        $r = $svc->checkout(['client_uuid' => $uuid(), 'items' => $cart, 'payments' => $pay, 'discount' => $disc], $kasir);
        $this->check('Kasir tanpa PIN: diskon ditolak', ! $r['success'] && str_contains($r['message'], 'otorisasi'), $r['message']);
        $r = $svc->checkout(['client_uuid' => $uuid(), 'items' => $cart, 'payments' => $pay, 'discount' => $disc, 'supervisor' => ['id' => $spvId, 'pin' => '9999']], $kasir);
        $this->check('Kasir + PIN salah: ditolak', ! $r['success'] && str_contains($r['message'], 'PIN salah'), $r['message']);
        $this->check('PIN salah di checkout menambah hitungan gagal (tidak ikut rollback)', (int) $db->table('users')->where('id', $spvId)->get()->getRow('pin_failed') === 1);
        $r = $svc->checkout(['client_uuid' => $uuid(), 'items' => $cart, 'payments' => $pay, 'discount' => ['type' => 'PCT', 'value' => 10, 'reason' => ''], 'supervisor' => ['id' => $spvId, 'pin' => '4821']], $kasir);
        $this->check('Diskon tanpa alasan ditolak', ! $r['success']);
        $r = $svc->checkout(['client_uuid' => $uuid(), 'items' => $cart, 'payments' => $pay, 'discount' => ['type' => 'PCT', 'value' => 100, 'reason' => 'x'], 'supervisor' => ['id' => $spvId, 'pin' => '4821']], $kasir);
        $this->check('Diskon 100% ditolak', ! $r['success']);
        $r = $svc->checkout(['client_uuid' => $uuid(), 'items' => $cart, 'payments' => $pay, 'discount' => ['type' => 'AMT', 'value' => 40000, 'reason' => 'x'], 'supervisor' => ['id' => $spvId, 'pin' => '4821']], $kasir);
        $this->check('Diskon nominal ≥ subtotal ditolak', ! $r['success']);
        $this->check('Penolakan tidak meninggalkan penjualan/stok', $db->table('pos_sales')->countAllResults() === 0);

        $ok = $svc->checkout(['client_uuid' => $uuid(), 'items' => $cart, 'payments' => $pay, 'discount' => $disc, 'supervisor' => ['id' => $spvId, 'pin' => '4821']], $kasir);
        $this->check('Kasir + PIN benar: diskon 10% berhasil, total 36.000, kembalian 4.000', $ok['success'] && $ok['grand_total'] == 36000.0 && $ok['change_amount'] == 4000.0, $ok);
        $sale = $db->table('pos_sales')->where('id', $ok['id'])->get()->getRowArray();
        $this->check('Header: subtotal 40.000, diskon 4.000, tipe PCT 10, alasan & penyetuju supervisor, kasir tetap kasir',
            (float) $sale['subtotal'] == 40000.0 && (float) $sale['discount_total'] == 4000.0 && $sale['discount_type'] === 'PCT' && (float) $sale['discount_value'] == 10.0
            && $sale['discount_reason'] === 'pelanggan langganan' && (int) $sale['discount_approved_by'] === $spvId && (int) $sale['cashier_id'] === $kasirId, $sale);
        $items = $db->table('pos_sale_items')->where('sale_id', $ok['id'])->orderBy('line_no')->get()->getResultArray();
        $this->check('Alokasi proporsional: baris A diskon 2.000 (18.000), baris B diskon 2.000 (18.000)', (float) $items[0]['discount'] == 2000.0 && (float) $items[0]['line_total'] == 18000.0 && (float) $items[1]['discount'] == 2000.0 && (float) $items[1]['line_total'] == 18000.0, array_column($items, 'discount'));
        $this->check('Σ line_total = grand_total', array_sum(array_column($items, 'line_total')) == (float) $sale['grand_total']);
        $this->check('Kas Kasir masuk 36.000 (nilai yang dibebankan)', (float) $db->query("SELECT COALESCE(SUM(amount),0) s FROM financial_transactions WHERE reference_type='POS_PAYMENT' AND id > 137")->getRow('s') == 36000.0);
        $this->check('HPP tidak berubah oleh diskon (2×5.000 + 4×2.000 = 18.000)', (float) $sale['cogs_total'] == 18000.0, $sale['cogs_total']);

        // diskon nominal dengan alokasi sisa
        $ok2 = $svc->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $A, 'qty' => 1], ['product_id' => $B, 'qty' => 1], ['product_id' => $B, 'qty' => 1]], 'payments' => [['method' => 'CASH', 'tendered' => 20000]], 'discount' => ['type' => 'AMT', 'value' => 1000, 'reason' => 'kemasan penyok'], 'supervisor' => ['id' => $spvId, 'pin' => '4821']], $kasir);
        $it2 = $db->table('pos_sale_items')->where('sale_id', $ok2['id'])->get()->getResultArray();
        $this->check('Diskon Rp1.000: total 19.000, Σ diskon baris = 1.000', $ok2['success'] && $ok2['grand_total'] == 19000.0 && array_sum(array_column($it2, 'discount')) == 1000.0, $ok2['message'] ?? '');

        // supervisor login langsung (tanpa PIN)
        $spvActor = ['user_id' => $spvId, 'can_override' => true, 'can_discount' => true];
        $ok3 = $svc->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $B, 'qty' => 2]], 'payments' => [['method' => 'CASH', 'tendered' => 9000]], 'discount' => ['type' => 'PCT', 'value' => 10, 'reason' => 'supervisor']], $spvActor);
        $this->check('Supervisor login: diskon tanpa PIN, penyetuju = dirinya', $ok3['success'] && (int) $db->table('pos_sales')->where('id', $ok3['id'])->get()->getRow('discount_approved_by') === $spvId, $ok3['message'] ?? '');

        // override stok lewat PIN: A sisa 5-2-1 = 2
        $o1 = $svc->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $A, 'qty' => 5]], 'payments' => [['method' => 'CASH', 'tendered' => 50000]]], $kasir);
        $this->check('Stok kurang: kasir ditolak', ! $o1['success']);
        $o2 = $svc->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $A, 'qty' => 5]], 'payments' => [['method' => 'CASH', 'tendered' => 50000]], 'override_reason' => 'penerimaan belum dicatat'], $kasir);
        $this->check('Override tanpa PIN ditolak', ! $o2['success']);
        $o3 = $svc->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $A, 'qty' => 5]], 'payments' => [['method' => 'CASH', 'tendered' => 50000]], 'override_reason' => 'penerimaan belum dicatat', 'supervisor' => ['id' => $spvId, 'pin' => '4821']], $kasir);
        $this->check('Override dengan PIN supervisor berhasil', $o3['success'], $o3['message'] ?? '');
        $ov = $db->table('pos_stock_overrides')->where('sale_id', $o3['id'] ?? 0)->get()->getRowArray();
        $this->check('Laporan pengecualian: penyetuju = supervisor (bukan kasir)', $ov && (int) $ov['supervisor_id'] === $spvId, $ov);

        // retur penjualan berdiskon: refund = harga yang dibayar
        $first = $items[0];
        $rt = (new PosReturnService())->create($ok['id'], ['items' => [['sale_item_id' => (int) $first['id'], 'qty' => 2, 'condition' => 'RESTOCK']], 'reason_code' => 'BATAL_BELI', 'refund_method' => 'CASH'], $spvId);
        $this->check('Retur baris berdiskon: refund 18.000 (bukan 20.000)', $rt['success'] && $rt['refund'] == 18000.0, $rt['message'] ?? '');

        // Bersihkan
        $db->query('SET FOREIGN_KEY_CHECKS=0');
        foreach (['pos_qris_settlements', 'pos_cash_deposits', 'stock_opening_lines', 'stock_openings', 'pos_payment_corrections', 'pos_return_items', 'pos_returns', 'stock_opname_counts', 'stock_broken_reports', 'stock_opname_lines', 'stock_opname_sessions', 'pos_daily_closings', 'pos_stock_overrides', 'pos_payments', 'pos_sale_items', 'pos_sales', 'stock_ledger', 'stock_balances', 'goods_receipt_items', 'goods_receipts', 'pos_markdowns', 'pos_product_prices'] as $tb) {
            $db->query("TRUNCATE TABLE $tb");
        }
        $db->query('SET FOREIGN_KEY_CHECKS=1');
        $db->query('DELETE FROM financial_transactions WHERE id > 137');
        $db->query('ALTER TABLE financial_transactions AUTO_INCREMENT = 138');
        $db->table('users')->whereIn('id', [$kasirId, $spvId, $spv2Id])->delete();
        $db->table('products')->update(['is_pos' => 0]);
        $db->table('activity_logs')->whereIn('module', ['pos', 'stock'])->delete();

        CLI::write($this->fails === 0 ? 'SEMUA LULUS' : $this->fails . ' GAGAL', $this->fails === 0 ? 'green' : 'red');
    }
}
