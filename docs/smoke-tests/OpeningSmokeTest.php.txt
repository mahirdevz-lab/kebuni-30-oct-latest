<?php

namespace App\Commands;

use App\Services\DailyClosingService;
use App\Services\PosSaleService;
use App\Services\StockLedgerService;
use App\Services\StockOpeningService;
use App\Services\StockOpnameService;
use CodeIgniter\CLI\BaseCommand;
use CodeIgniter\CLI\CLI;

/** TEMPORARY: stok awal (OPENING). Membersihkan data uji sendiri. */
class OpeningSmokeTest extends BaseCommand
{
    protected $group = 'pos';
    protected $name  = 'pos:opening-smoke';

    private int $fails = 0;

    private function check(string $label, bool $ok, $detail = ''): void
    {
        if (! $ok) {
            $this->fails++;
        }
        CLI::write(($ok ? '[PASS] ' : '[FAIL] ') . $label . ($detail !== '' ? '  => ' . (is_scalar($detail) ? $detail : json_encode($detail)) : ''), $ok ? 'green' : 'red');
    }

    public function run(array $params)
    {
        $db  = db_connect();
        $ids = array_map('intval', array_column($db->table('products')->where('status', 'aktif')->where('deleted_at', null)->orderBy('id')->get(4)->getResultArray(), 'id'));
        [$A, $B, $C, $D] = $ids;
        $svc = new StockOpeningService(); $led = new StockLedgerService();
        $cut = date('Y-m-d H:i:s', time() - 3600);

        $this->check('Cut-off di masa depan ditolak', ! $svc->createDraft(date('Y-m-d H:i:s', time() + 3600), null, 1)['success']);
        $d1 = $svc->createDraft($cut, 'go-live uji', 1);
        $this->check('Draft dibuat', $d1['success'], $d1['message']);
        $this->check('Draft kedua ditolak selama masih ada draft', ! $svc->createDraft($cut, null, 1)['success']);

        $dc = $svc->defaultCost(2);
        $this->check('Biaya default = harga beli price period terbaru', $dc['cost'] == 25000.0 && str_contains((string) $dc['source'], 'PP-2026-09-19'), $dc);

        $this->check('Produk dobel ditolak', ! $svc->saveLines($d1['id'], [['product_id' => $A, 'qty' => 1, 'unit_cost' => 1], ['product_id' => $A, 'qty' => 2, 'unit_cost' => 1]])['success']);
        $this->check('Qty negatif ditolak', ! $svc->saveLines($d1['id'], [['product_id' => $A, 'qty' => -1, 'unit_cost' => 1]])['success']);
        $s = $svc->saveLines($d1['id'], [
            ['product_id' => $A, 'qty' => '120,5', 'unit_cost' => '10000'],
            ['product_id' => $B, 'qty' => '0', 'unit_cost' => '5000'],
            ['product_id' => $C, 'qty' => '5', 'unit_cost' => '0'],
        ]);
        $this->check('Simpan 3 baris (koma desimal diterima)', $s['success'] && (float) $db->table('stock_opening_lines')->where('product_id', $A)->get()->getRow('qty') === 120.5, $s['message']);
        $p = $svc->post($d1['id'], 1);
        $this->check('Posting ditolak bila biaya 0', ! $p['success'] && str_contains($p['message'], 'Biaya'), $p['message']);
        $svc->saveLines($d1['id'], [['product_id' => $A, 'qty' => '120,5', 'unit_cost' => '10000'], ['product_id' => $B, 'qty' => '0', 'unit_cost' => '5000'], ['product_id' => $C, 'qty' => '5', 'unit_cost' => '8000']]);
        $p = $svc->post($d1['id'], 1);
        $this->check('Posting berhasil (2 produk, nilai 1.245.000)', $p['success'] && str_contains($p['message'], '1.245.000'), $p['message']);
        $bA = $led->balance($A, 1);
        $this->check('Saldo A 120,5 @ 10.000', (float) $bA['qty'] === 120.5 && (float) $bA['avg_cost'] == 10000.0, $bA);
        $lg = $db->table('stock_ledger')->where('movement_type', 'OPENING')->get()->getResultArray();
        $this->check('Ledger OPENING 2 baris, waktu efektif = cut-off', count($lg) === 2 && $lg[0]['effective_at'] === $cut, $lg[0]['effective_at'] ?? null);
        $this->check('Baris qty 0 tidak diposting & dihapus', ! $db->table('stock_opening_lines')->where('product_id', $B)->countAllResults());
        $this->check('Produk berstok awal ditandai POS', (int) $db->table('products')->where('id', $A)->get()->getRow('is_pos') === 1 && (int) $db->table('products')->where('id', $C)->get()->getRow('is_pos') === 1);
        $this->check('Dokumen POSTED tidak bisa diubah', ! $svc->saveLines($d1['id'], [])['success'] && ! $svc->post($d1['id'], 1)['success']);

        // Stok awal kedua untuk produk yang sudah punya mutasi ditolak
        $d2 = $svc->createDraft($cut, null, 1);
        $svc->saveLines($d2['id'], [['product_id' => $A, 'qty' => 1, 'unit_cost' => 1], ['product_id' => $D, 'qty' => 10, 'unit_cost' => 3000]]);
        $p2 = $svc->post($d2['id'], 1);
        $this->check('Produk yang sudah ada mutasi ditolak', ! $p2['success'] && str_contains($p2['message'], 'sudah punya mutasi'), $p2['message']);
        $svc->saveLines($d2['id'], [['product_id' => $D, 'qty' => 10, 'unit_cost' => 3000]]);

        // Posting ditolak selama opname dihitung
        $op = new StockOpnameService();
        $o  = $op->open(1);
        $this->check('Posting ditolak selama opname dihitung', $o['success'] && ! $svc->post($d2['id'], 1)['success'], $o['message']);
        $op->cancel($o['id'], 'uji');
        $this->check('Setelah opname dibatalkan, posting D berhasil', $svc->post($d2['id'], 1)['success'] && (float) $led->balance($D, 1)['qty'] === 10.0);

        // Penjualan setelah stok awal memakai biaya stok awal sebagai HPP
        $db->table('pos_product_prices')->insert(['product_id' => $A, 'retail_price' => 15000, 'effective_from' => date('Y-m-d H:i:s', time() - 60), 'created_at' => date('Y-m-d H:i:s')]);
        $day = new DailyClosingService(); $day->open(0, 1);
        $sale = (new PosSaleService())->checkout(['client_uuid' => '22222222-3333-4444-8555-666666666666', 'items' => [['product_id' => $A, 'qty' => 2]], 'payments' => [['method' => 'CASH', 'tendered' => 30000]]], ['user_id' => 1, 'can_override' => false]);
        $this->check('HPP penjualan = 2 × 10.000 dari stok awal', $sale['success'] && (float) $db->table('pos_sales')->where('id', $sale['id'])->get()->getRow('cogs_total') == 20000.0);
        $sum = (float) $db->query('SELECT COALESCE(SUM(qty_in-qty_out),0) s FROM stock_ledger WHERE product_id=?', [$A])->getRow('s');
        $this->check('Saldo = Σ ledger (A 118,5)', $sum === 118.5 && (float) $led->balance($A, 1)['qty'] === 118.5, $sum);

        // Bersihkan
        $db->query('SET FOREIGN_KEY_CHECKS=0');
        foreach (['pos_qris_settlements', 'pos_cash_deposits', 'stock_opening_lines', 'stock_openings', 'pos_payment_corrections', 'pos_return_items', 'pos_returns', 'stock_opname_counts', 'stock_broken_reports', 'stock_opname_lines', 'stock_opname_sessions', 'pos_daily_closings', 'pos_stock_overrides', 'pos_payments', 'pos_sale_items', 'pos_sales', 'stock_ledger', 'stock_balances', 'goods_receipt_items', 'goods_receipts', 'pos_markdowns', 'pos_product_prices'] as $tb) {
            $db->query("TRUNCATE TABLE $tb");
        }
        $db->query('SET FOREIGN_KEY_CHECKS=1');
        $db->query('DELETE FROM financial_transactions WHERE id > 137');
        $db->query('ALTER TABLE financial_transactions AUTO_INCREMENT = 138');
        $db->table('products')->update(['is_pos' => 0]);
        $db->table('activity_logs')->whereIn('module', ['pos', 'stock'])->delete();

        CLI::write($this->fails === 0 ? 'SEMUA LULUS' : $this->fails . ' GAGAL', $this->fails === 0 ? 'green' : 'red');
    }
}
