<?php

namespace App\Commands;

use App\Services\DailyClosingService;
use App\Services\GoodsReceiptService;
use App\Services\PosReturnService;
use App\Services\PosSaleService;
use App\Services\ProfitShrinkageService;
use App\Services\StockLedgerService;
use CodeIgniter\CLI\BaseCommand;
use CodeIgniter\CLI\CLI;

/** TEMPORARY: retur pelanggan & koreksi metode bayar setelah EOD. Membersihkan data uji sendiri. */
class ReturnSmokeTest extends BaseCommand
{
    protected $group       = 'pos';
    protected $name        = 'pos:return-smoke';
    protected $description = 'Smoke test retur (sementara)';

    private int $fails = 0;

    private function check(string $label, bool $ok, $detail = ''): void
    {
        if (! $ok) {
            $this->fails++;
        }
        CLI::write(($ok ? '[PASS] ' : '[FAIL] ') . $label . ($detail !== '' ? '  => ' . (is_scalar($detail) ? $detail : json_encode($detail)) : ''), $ok ? 'green' : 'red');
    }

    public function run(array $params)
    {
        $db  = db_connect();
        $ids = array_map('intval', array_column($db->table('products')->where('status', 'aktif')->where('deleted_at', null)->orderBy('id')->get(2)->getResultArray(), 'id'));
        [$A, $B] = $ids;
        $sup = (int) $db->table('suppliers')->where('deleted_at', null)->get(1)->getRow('id');
        $cust = (int) $db->table('accounts')->where('deleted_at', null)->get(1)->getRow('id');
        $db->table('products')->whereIn('id', $ids)->update(['is_pos' => 1]);
        foreach ([$A => 15000, $B => 6000] as $pid => $price) {
            $db->table('pos_product_prices')->insert(['product_id' => $pid, 'retail_price' => $price, 'effective_from' => date('Y-m-d H:i:s', time() - 60), 'created_at' => date('Y-m-d H:i:s')]);
        }
        $loc  = 1;
        $led  = new StockLedgerService(); $day = new DailyClosingService(); $sale = new PosSaleService(); $ret = new PosReturnService();
        $qty  = static fn (int $p) => (float) $led->balance($p, $loc)['qty'];
        $acc  = static fn (int $id, ?string $date = null) => (float) db_connect()->query("SELECT COALESCE(SUM(CASE WHEN type='INCOME' THEN amount ELSE -amount END),0) s FROM financial_transactions WHERE cash_account_id=? AND id > 137" . ($date ? ' AND transaction_date = ?' : ''), $date ? [$id, $date] : [$id])->getRow('s');
        $kasir = ['user_id' => 1, 'can_override' => false];
        $uuid  = static fn () => sprintf('%04x%04x-%04x-%04x-%04x-%04x%04x%04x', mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff));
        $itemOf = static fn (int $saleId, int $pid, string $type = 'NORMAL') => (int) db_connect()->table('pos_sale_items')->where('sale_id', $saleId)->where('product_id', $pid)->where('line_type', $type)->get()->getRow('id');

        // Hari 1
        $day->open(100000, 1);
        $d1 = $day->openDay();
        (new GoodsReceiptService())->create(['supplier_id' => $sup, 'items' => [['product_id' => $A, 'qty_regular' => 100, 'line_cost' => 1000000], ['product_id' => $B, 'qty_regular' => 50, 'line_cost' => 200000]]], 1);
        $s1 = $sale->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $A, 'qty' => 10], ['product_id' => $B, 'qty' => 5, 'free_qty' => 1]], 'payments' => [['method' => 'CASH', 'tendered' => 200000]]], $kasir);
        $s2 = $sale->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $A, 'qty' => 2]], 'payments' => [['method' => 'QRIS']]], $kasir);
        $s3 = $sale->checkout(['client_uuid' => $uuid(), 'customer_id' => $cust, 'items' => [['product_id' => $A, 'qty' => 4]], 'payments' => [['method' => 'CASH', 'tendered' => 20000]]], $kasir);
        $this->check('Setup: 3 penjualan (tunai, QRIS pending, sebagian)', $s1['success'] && $s2['success'] && $s3['success'] && $s3['payment_status'] === 'PARTIAL');
        $sum1 = $day->summary($d1);
        $day->close((int) $d1['id'], $sum1['expected_cash'], 1);

        // Setelah EOD, toko tutup
        $r = $ret->create($s1['id'], ['items' => [['sale_item_id' => $itemOf($s1['id'], $A), 'qty' => 2, 'condition' => 'RESTOCK']], 'reason_code' => 'TIDAK_SESUAI', 'refund_method' => 'CASH'], 1);
        $this->check('Refund tunai saat toko tutup ditolak', ! $r['success'], $r['message']);
        $this->check('Void penjualan hari yang sudah EOD ditolak', ! $sale->void($s1['id'], 'x', 1)['success']);
        $bank0 = $acc(1);
        $r1 = $ret->create($s1['id'], ['items' => [['sale_item_id' => $itemOf($s1['id'], $A), 'qty' => 2, 'condition' => 'RESTOCK']], 'reason_code' => 'TIDAK_SESUAI', 'refund_method' => 'TRANSFER', 'cash_account_id' => 1], 1);
        $this->check('Retur setelah EOD via transfer Bank Utama', $r1['success'] && $r1['refund'] == 30000.0, $r1['message']);
        $this->check('Stok A kembali +2 (100-16+2 = 86)', $qty($A) === 86.0, $qty($A));
        $this->check('Bank Utama -30.000', $acc(1) - $bank0 == -30000.0, $acc(1) - $bank0);
        $this->check('Retur melebihi sisa ditolak', ! $ret->create($s1['id'], ['items' => [['sale_item_id' => $itemOf($s1['id'], $A), 'qty' => 9, 'condition' => 'RESTOCK']], 'reason_code' => 'BATAL_BELI', 'refund_method' => 'TRANSFER', 'cash_account_id' => 1], 1)['success']);
        $this->check('Retur penjualan dengan QRIS pending ditolak', ! $ret->create($s2['id'], ['items' => [['sale_item_id' => $itemOf($s2['id'], $A), 'qty' => 1, 'condition' => 'RESTOCK']], 'reason_code' => 'BATAL_BELI', 'refund_method' => 'TRANSFER', 'cash_account_id' => 1], 1)['success']);
        $this->check('Alasan LAINNYA tanpa keterangan ditolak', ! $ret->create($s1['id'], ['items' => [['sale_item_id' => $itemOf($s1['id'], $A), 'qty' => 1, 'condition' => 'RESTOCK']], 'reason_code' => 'LAINNYA', 'refund_method' => 'TRANSFER', 'cash_account_id' => 1], 1)['success']);

        // Hari 2 (tanggal bisnis besok)
        $biz2 = date('Y-m-d', strtotime('+1 day'));
        $db->table('pos_daily_closings')->insert(['location_id' => 1, 'business_date' => $biz2, 'status' => 'OPEN', 'opening_cash' => 200000, 'opened_at' => date('Y-m-d H:i:s'), 'created_at' => date('Y-m-d H:i:s')]);
        $kas0 = $acc(3, $biz2);
        $r2 = $ret->create($s1['id'], ['items' => [['sale_item_id' => $itemOf($s1['id'], $A), 'qty' => 3, 'condition' => 'DAMAGED']], 'reason_code' => 'RUSAK', 'reason' => 'busuk di dalam', 'refund_method' => 'CASH'], 1);
        $this->check('Retur rusak, refund tunai di hari 2', $r2['success'] && $r2['refund'] == 45000.0, $r2['message']);
        $this->check('Kas Kasir hari 2 -45.000 (tanggal bisnis hari 2)', $acc(3, $biz2) - $kas0 == -45000.0);
        $this->check('Stok A netto tetap 86 (masuk 3 lalu Broken 3)', $qty($A) === 86.0, $qty($A));
        $bs = $db->table('stock_broken_reports')->where('status', 'POSTED')->get()->getRowArray();
        $this->check('Broken Stock otomatis POSTED nilai 30.000', $bs && (float) $bs['posted_value'] == 30000.0 && str_contains($bs['reason'], 'Retur'), $bs['posted_value'] ?? null);
        $d2 = $day->openDay();
        $sum2 = $day->summary($d2);
        $this->check('EOD hari 2: refund 45.000, kas seharusnya 155.000', $sum2['cash_refunds'] == 45000.0 && $sum2['expected_cash'] == 155000.0, [$sum2['cash_refunds'], $sum2['expected_cash']]);

        $r3 = $ret->create($s1['id'], ['items' => [['sale_item_id' => $itemOf($s1['id'], $B, 'PROMO_FREE'), 'qty' => 1, 'condition' => 'RESTOCK']], 'reason_code' => 'BATAL_BELI', 'refund_method' => 'CASH'], 1);
        $row3 = $db->table('pos_returns')->where('id', $r3['id'] ?? 0)->get()->getRowArray();
        $this->check('Retur barang gratis promo: nilai 0, tanpa refund (NONE)', $r3['success'] && (float) $row3['return_value'] == 0.0 && $row3['refund_method'] === 'NONE' && $qty($B) === 45.0, [$row3['refund_method'] ?? null, $qty($B)]);

        $r4 = $ret->create($s3['id'], ['items' => [['sale_item_id' => $itemOf($s3['id'], $A), 'qty' => 4, 'condition' => 'RESTOCK']], 'reason_code' => 'BATAL_BELI', 'refund_method' => 'CASH'], 1);
        $s3row = $db->table('pos_sales')->where('id', $s3['id'])->get()->getRowArray();
        $this->check('Retur penjualan belum lunas: refund 20.000, sisa tagihan 0, lunas', $r4['success'] && $r4['refund'] == 20000.0 && (float) $s3row['balance_due'] == 0.0 && $s3row['payment_status'] === 'PAID', [$r4['refund'] ?? null, $s3row['balance_due'], $s3row['payment_status']]);

        $s4 = $sale->checkout(['client_uuid' => $uuid(), 'items' => [['product_id' => $A, 'qty' => 1]], 'payments' => [['method' => 'CASH', 'tendered' => 15000]]], $kasir);
        $ret->create($s4['id'], ['items' => [['sale_item_id' => $itemOf($s4['id'], $A), 'qty' => 1, 'condition' => 'RESTOCK']], 'reason_code' => 'SALAH_PRODUK', 'refund_method' => 'CASH'], 1);
        $v = $sale->void($s4['id'], 'uji', 1);
        $this->check('Void ditolak bila sudah ada retur (cegah refund ganda)', ! $v['success'], $v['message']);

        // Koreksi metode bayar (hari 1, sudah EOD)
        $p1 = $db->table('pos_payments')->where('sale_id', $s1['id'])->get()->getRowArray();
        $d1date = $d1['business_date'];
        $kasD1 = $acc(3, $d1date); $qrD1 = $acc(4, $d1date);
        $c = $ret->correctPayment((int) $p1['id'], 'QRIS', 'pelanggan bayar QRIS, kasir salah pilih', 1);
        $this->check('Koreksi tunai → QRIS pada tanggal bisnis hari 1', $c['success'] && $acc(3, $d1date) - $kasD1 == -180000.0 && $acc(4, $d1date) - $qrD1 == 180000.0, $c['message']);
        $this->check('Snapshot EOD hari 1 tidak berubah', (float) $db->table('pos_daily_closings')->where('id', $d1['id'])->get()->getRow('expected_cash') == (float) $sum1['expected_cash']);
        $this->check('Koreksi kedua ditolak', ! $ret->correctPayment((int) $p1['id'], 'CASH', 'x', 1)['success']);
        $p2 = $db->table('pos_payments')->where('sale_id', $s2['id'])->get()->getRowArray();
        $this->check('Koreksi pembayaran QRIS pending ditolak', ! $ret->correctPayment((int) $p2['id'], 'CASH', 'x', 1)['success']);
        $this->check('Koreksi tidak masuk laporan pemasukan/pengeluaran (kategori Pemindahan Dana)', (int) $db->query("SELECT COUNT(*) c FROM financial_transactions ft JOIN financial_categories fc ON fc.id = ft.category_id WHERE ft.reference_type IN ('TRANSFER_OUT','TRANSFER_IN') AND ft.id > 137 AND fc.name = 'Pemindahan Dana'")->getRow('c') === 2);

        // Laporan
        $an = (new ProfitShrinkageService())->build(date('Y-m-d'), $biz2);
        $this->check('Laba rugi: penjualan bersih 135.000 (285.000 - retur 150.000), known 30.000', $an['pl']['revenue'] == 135000.0 && $an['pl']['returns'] == 150000.0 && $an['pl']['known'] == 30000.0, $an['pl']);
        foreach ([$A => 90.0, $B => 45.0] as $pid => $exp) {
            $sum = (float) $db->query('SELECT COALESCE(SUM(qty_in-qty_out),0) s FROM stock_ledger WHERE product_id=?', [$pid])->getRow('s');
            $this->check("Saldo = Σ ledger (produk $pid = $exp)", $sum === $exp && $qty($pid) === $exp, [$sum, $qty($pid)]);
        }

        // Bersihkan
        $db->query('SET FOREIGN_KEY_CHECKS=0');
        foreach (['pos_qris_settlements', 'pos_cash_deposits', 'stock_opening_lines', 'stock_openings', 'pos_payment_corrections', 'pos_return_items', 'pos_returns', 'stock_opname_counts', 'stock_broken_reports', 'stock_opname_lines', 'stock_opname_sessions', 'pos_daily_closings', 'pos_stock_overrides', 'pos_payments', 'pos_sale_items', 'pos_sales', 'stock_ledger', 'stock_balances', 'goods_receipt_items', 'goods_receipts', 'pos_markdowns', 'pos_product_prices'] as $tb) {
            $db->query("TRUNCATE TABLE $tb");
        }
        $db->query('SET FOREIGN_KEY_CHECKS=1');
        $db->query('DELETE FROM financial_transactions WHERE id > 137');
        $db->query('ALTER TABLE financial_transactions AUTO_INCREMENT = 138');
        $db->table('products')->whereIn('id', $ids)->update(['is_pos' => 0]);
        $db->table('activity_logs')->whereIn('module', ['pos', 'stock'])->delete();

        CLI::write($this->fails === 0 ? 'SEMUA LULUS' : $this->fails . ' GAGAL', $this->fails === 0 ? 'green' : 'red');
    }
}
